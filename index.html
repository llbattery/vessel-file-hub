const FOLDERS = { "Vessel Files": "1aZxiuvh2vozEhBYWWZYROwS9VwoyY-A-", "Voyage Files": "18jTuqx2uF9pR7fFCwJTg43a141f34bEh", "Miscellaneous": "1kMFh_47hTqDXtQb_VCexcx-w4RJZyk0K" }; function doGet(e) { try { var action = e.parameter.action; if (action === "search") { var query = e.parameter.q ? e.parameter.q.toLowerCase().trim() : ""; var category = e.parameter.category || "All"; if (category === "Voyage Plans") { category = "Voyage Files"; } var results = []; var foldersToSearch = []; if (category === "All" || !FOLDERS[category]) { for (var cat in FOLDERS) { foldersToSearch.push({name: cat, id: FOLDERS[cat]}); } } else { foldersToSearch.push({name: category, id: FOLDERS[category]}); } foldersToSearch.forEach(function(folderInfo) { try { var mainFolder = DriveApp.getFolderById(folderInfo.id); searchFolderRecursive(mainFolder, query, folderInfo.name, results); } catch (err) { results.push({ name: "CRITICAL ERROR: " + err.message, category: folderInfo.name, downloadUrl: "#", viewUrl: "#" }); } }); // TRAP: If query was empty and no files found at all, report a helpful notice if (results.length === 0) { results.push({ name: "NOTICE: Folder scanned successfully, but 0 files were found inside. Check if files are uploaded in subfolders!", category: category, downloadUrl: "#", viewUrl: "#" }); } return ContentService .createTextOutput(JSON.stringify(results)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status: "Active"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({error: err.message})) .setMimeType(ContentService.MimeType.JSON); } } function searchFolderRecursive(folder, query, categoryName, results) { try { var files = folder.getFiles(); while (files.hasNext()) { var file = files.next(); var fileName = file.getName(); var fileId = file.getId(); var searchName = fileName.toLowerCase(); if (categoryName === "Vessel Files") { searchName = searchName .replace(/\s+data\.car$/, "") .replace(/\.car$/, "") .trim(); } if (categoryName === "Voyage Files") { searchName = searchName .replace(/\.xls$/, "") .trim(); } // If query is empty, it bypasses filtering and accepts ALL files if (query === "" || searchName.indexOf(query) !== -1) { results.push({ name: fileName, category: categoryName, downloadUrl: file.getUrl(), viewUrl: file.getUrl() }); } } } catch (fileErr) { // Skip file iteration errors safely } try { var subfolders = folder.getFolders(); while (subfolders.hasNext()) { var subfolder = subfolders.next(); searchFolderRecursive(subfolder, query, categoryName, results); } } catch (subErr) { // Skip subfolder traversal errors safely } }
